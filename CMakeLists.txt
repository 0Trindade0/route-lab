cmake_minimum_required(VERSION 3.16)
project(route_lab VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(route_lab src/main.cpp)
target_include_directories(route_lab PRIVATE include)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_graph tests/test_graph.cpp)
  target_include_directories(test_graph PRIVATE include)
  add_test(NAME graph_behaviors COMMAND test_graph)
endif()
